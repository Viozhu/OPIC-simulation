# Fase 3 — Interfaz web y evaluación en paralelo: plan de implementación

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Que un candidato haga el examen completo en el navegador y reciba su nivel ACTFL, con la evaluación corriendo en paralelo al propio examen.

**Architecture:** Dos capas nuevas sobre lo que ya existe. `session/` es la sesión como dato: un `session.json` por carpeta, una máquina de estados que rechaza transiciones ilegales, y un worker de un solo hilo que consume una cola de tareas (generar preguntas, sintetizar audio, evaluar una respuesta, agregar). `web/` es sólo HTTP: FastAPI valida y delega, y sirve una única página cuyo JavaScript gobierna los cinco pasos. `core/` no se toca y sigue sin importar nada de esto.

**Tech Stack:** Python 3.13 con uv, FastAPI + uvicorn, HTML y módulos ES nativos sin paso de build, pytest. Ollama (`qwen3:14b`), faster-whisper, `say`.

**Spec:** `docs/superpowers/specs/2026-08-19-phase-3-web-design.md`

## Global Constraints

- `core/` no importa `fastapi`, ni `opic.adapters`, ni `opic.session`. Lo comprueba `tests/test_core_purity.py`.
- `session/` puede importar `core`. Sólo `session/worker.py` toca los adaptadores de IA.
- `web/app.py` no contiene lógica de examen: valida, delega en `session/`, devuelve JSON.
- Todo texto que ve el usuario final va en coreano. Los identificadores, nombres de fichero, mensajes de commit y comentarios de código, en inglés.
- Los textos coreanos del survey y del self assessment se sirven desde `core/`; el HTML no los duplica.
- Los tests nuevos corren sin Ollama, sin Whisper, sin `say` y sin navegador. Lo que necesite un modelo real va tras `@pytest.mark.models`.
- `session.json` se escribe entero a un temporal y se publica con `Path.replace`. Nunca se escribe en el sitio.
- Un solo worker de uvicorn. Dos procesos serían dos colas peleando por el mismo modelo.
- Sin dependencias de frontend. `<script type="module">` y nada más.
- Cada tarea termina con `uv run --extra dev pytest` en verde antes del commit.

```python
import time
from pathlib import Path

import pytest

from opic.adapters.llm import OllamaClient
from opic.core.questions import EXAM_LENGTH, build_blueprint
from opic.core.survey import Housing, Occupation, RecentCourse, SurveyAnswers

TWELVE_ITEMS = (
    "movies", "clubbing", "live_shows", "concerts", "museums", "parks",
    "listening_music", "cooking", "yoga", "walking", "domestic_travel", "overseas_travel",
)


@pytest.mark.models
def test_write_full_question_set_timing(tmp_path: Path) -> None:
    answers = SurveyAnswers(
        occupation=Occupation.BUSINESS,
        is_student=False,
        housing=Housing.ALONE,
        selected_item_ids=TWELVE_ITEMS,
        recent_course=RecentCourse.LANGUAGE,
    )
    blueprint = build_blueprint(answers, self_assess_level=4, seed=7)

    elapsed = []
    for trial in range(3):
        started = time.monotonic()
        questions = OllamaClient().write(answers, blueprint)
        elapsed.append(time.monotonic() - started)
        assert len(questions) == EXAM_LENGTH

    (tmp_path / "timing.json").write_text(json.dumps(elapsed), encoding="utf-8")
    print(f"\nwrite() elapsed: {[round(e, 1) for e in elapsed]}")
```

```python
# src/opic/session/state.py
"""A session as data, and the transitions it is allowed to make.

The browser proposes the next step; this module decides whether it is legal.
Nothing here touches disk, models or HTTP, so the whole state machine is
exercised in milliseconds.
"""

from __future__ import annotations

from dataclasses import dataclass, replace
from enum import Enum

from opic.core.survey import SurveyAnswers


class Step(Enum):
    SURVEY = "survey"
    LEVEL = "level"
    SETUP = "setup"
    SAMPLE = "sample"
    TEST = "test"
    GRADING = "grading"
    DONE = "done"


class AnswerState(Enum):
    PENDING = "pending"
    RECORDED = "recorded"
    TRANSCRIBED = "transcribed"
    SCORED = "scored"
    FAILED = "failed"


ORDER: tuple[Step, ...] = (
    Step.SURVEY,
    Step.LEVEL,
    Step.SETUP,
    Step.SAMPLE,
    Step.TEST,
    Step.GRADING,
    Step.DONE,
)


class TransitionError(ValueError):
    """Raised when the browser asks for a step that does not follow this one."""


@dataclass(frozen=True, slots=True)
class QuestionRecord:
    """A question as it is persisted: the slot flattened, plus its wording.

    The slot's enums are stored by value so `session.json` survives a rename in
    the code, and `audio_path` points into the TTS cache — the session never
    owns a copy of the audio.
    """

    index: int
    qtype: str
    topic_id: str | None
    combo_id: int | None
    advanced: bool
    target: str
    text_en: str
    audio_path: str | None


@dataclass(frozen=True, slots=True)
class AnswerRecord:
    index: int
    state: AnswerState
    audio_path: str | None = None
    error: str | None = None


@dataclass(frozen=True, slots=True)
class SessionState:
    id: str
    step: Step
    seed: int
    survey: SurveyAnswers | None
    self_assess_level: int | None
    questions: tuple[QuestionRecord, ...]
    answers: tuple[AnswerRecord, ...]

    @classmethod
    def new(cls, session_id: str, seed: int) -> SessionState:
        return cls(
            id=session_id,
            step=Step.SURVEY,
            seed=seed,
            survey=None,
            self_assess_level=None,
            questions=(),
            answers=(),
        )

    @property
    def questions_ready(self) -> bool:
        """Every question written and every audio synthesised."""
        return bool(self.questions) and all(q.audio_path for q in self.questions)

    def with_survey(self, survey: SurveyAnswers) -> SessionState:
        return replace(self, survey=survey)

    def with_level(self, level: int) -> SessionState:
        return replace(self, self_assess_level=level)

    def with_questions(self, questions: tuple[QuestionRecord, ...]) -> SessionState:
        answers = tuple(
            AnswerRecord(index=q.index, state=AnswerState.PENDING) for q in questions
        )
        return replace(self, questions=questions, answers=answers)

    def with_question_audio(self, index: int, audio_path: str) -> SessionState:
        questions = tuple(
            replace(q, audio_path=audio_path) if q.index == index else q
            for q in self.questions
        )
        return replace(self, questions=questions)

    def with_answer(self, answer: AnswerRecord) -> SessionState:
        answers = tuple(
            answer if existing.index == answer.index else existing
            for existing in self.answers
        )
        return replace(self, answers=answers)

    def pending_indices(self) -> tuple[int, ...]:
        return tuple(a.index for a in self.answers if a.state is AnswerState.PENDING)

    def scored_indices(self) -> tuple[int, ...]:
        return tuple(a.index for a in self.answers if a.state is AnswerState.SCORED)


def advance(state: SessionState, step: Step) -> SessionState:
    """Move to the next step, refusing anything that is not exactly next."""
    current = ORDER.index(state.step)
    if current == len(ORDER) - 1:
        raise TransitionError(f"{state.step.value} is the last step; nothing follows it")
    if ORDER.index(step) != current + 1:
        raise TransitionError(
            f"cannot go from {state.step.value} to {step.value}; "
            f"expected {ORDER[current + 1].value}"
        )
    return replace(state, step=step)
```

```python
# src/opic/session/store.py
"""The session on disk: one folder, one `session.json`, written by rename.

`session.json` is the source of truth. It is written whole to a temporary file
and published with `Path.replace`, so a crash mid-write leaves the previous
version intact rather than a truncated one.
"""

from __future__ import annotations

import json
from datetime import datetime
from pathlib import Path

from opic.core.survey import Housing, Occupation, RecentCourse, SurveyAnswers
from opic.session.state import (
    AnswerRecord,
    AnswerState,
    QuestionRecord,
    SessionState,
    Step,
)

SESSION_FILE = "session.json"
RESULT_FILE = "result.json"


def new_session_id(now: datetime) -> str:
    """Readable and sortable, with no new dependency."""
    return now.strftime("%Y%m%d-%H%M%S")


def session_dir(root: Path, session_id: str) -> Path:
    return root / session_id


def answer_audio_path(root: Path, session_id: str, index: int, suffix: str) -> Path:
    """Where one recorded answer lives. The suffix comes from the browser."""
    return session_dir(root, session_id) / f"answer-{index:02d}{suffix}"


def answer_result_path(root: Path, session_id: str, index: int) -> Path:
    return session_dir(root, session_id) / f"answer-{index:02d}.json"


def result_path(root: Path, session_id: str) -> Path:
    return session_dir(root, session_id) / RESULT_FILE


def create(root: Path, seed: int, now: datetime) -> SessionState:
    state = SessionState.new(new_session_id(now), seed)
    session_dir(root, state.id).mkdir(parents=True, exist_ok=True)
    save(root, state)
    return state


def save(root: Path, state: SessionState) -> None:
    directory = session_dir(root, state.id)
    directory.mkdir(parents=True, exist_ok=True)
    payload = json.dumps(to_json(state), ensure_ascii=False, indent=2)
    temporary = directory / f"{SESSION_FILE}.tmp"
    temporary.write_text(payload, encoding="utf-8")
    temporary.replace(directory / SESSION_FILE)


def load(root: Path, session_id: str) -> SessionState:
    path = session_dir(root, session_id) / SESSION_FILE
    if not path.exists():
        raise FileNotFoundError(f"no session {session_id!r} under {root}")
    return from_json(json.loads(path.read_text(encoding="utf-8")))


def list_unfinished(root: Path) -> tuple[str, ...]:
    """Sessions with a `session.json` and no `result.json`, oldest first."""
    if not root.exists():
        return ()
    return tuple(
        sorted(
            directory.name
            for directory in root.iterdir()
            if (directory / SESSION_FILE).exists()
            and not (directory / RESULT_FILE).exists()
        )
    )


def to_json(state: SessionState) -> dict:
    return {
        "id": state.id,
        "step": state.step.value,
        "seed": state.seed,
        "self_assess_level": state.self_assess_level,
        "survey": _survey_json(state.survey),
        "questions": [
            {
                "index": q.index,
                "qtype": q.qtype,
                "topic_id": q.topic_id,
                "combo_id": q.combo_id,
                "advanced": q.advanced,
                "target": q.target,
                "text_en": q.text_en,
                "audio_path": q.audio_path,
            }
            for q in state.questions
        ],
        "answers": [
            {
                "index": a.index,
                "state": a.state.value,
                "audio_path": a.audio_path,
                "error": a.error,
            }
            for a in state.answers
        ],
    }


def from_json(payload: dict) -> SessionState:
    return SessionState(
        id=payload["id"],
        step=Step(payload["step"]),
```

```python
# src/opic/session/assessments.py
"""One judged answer on disk, and the aggregation of whatever survived.

`answer-NN.json` holds everything needed to rebuild an `AnswerAssessment`:
words with their timings, the voice-activity spans, the metrics and the
dimension scores. That is what makes a restart able to aggregate without
re-running a single model.
"""

from __future__ import annotations

import json
from collections.abc import Sequence
from pathlib import Path

from opic.core.metrics import AnswerMetrics
from opic.core.questions import Question, QuestionSlot, QuestionType
from opic.core.rubric import ActflLevel, Dimension
from opic.core.scoring import AnswerAssessment, DimensionScore, Evidence, ExamResult, aggregate
from opic.core.transcript import Span, Transcript, Word
from opic.session.state import QuestionRecord
from opic.session.store import answer_result_path, session_dir


def question_from_record(record: QuestionRecord) -> Question:
    return Question(
        slot=QuestionSlot(
            index=record.index,
            qtype=QuestionType(record.qtype),
            topic_id=record.topic_id,
            combo_id=record.combo_id,
            advanced=record.advanced,
            target=ActflLevel[record.target],
        ),
        text_en=record.text_en,
    )


def write_answer(root: Path, session_id: str, assessment: AnswerAssessment) -> Path:
    path = answer_result_path(root, session_id, assessment.question_index)
    path.parent.mkdir(parents=True, exist_ok=True)
    temporary = path.with_suffix(".json.tmp")
    temporary.write_text(
        json.dumps(_answer_json(assessment), ensure_ascii=False, indent=2),
        encoding="utf-8",
    )
    temporary.replace(path)
    return path


def read_answer(root: Path, session_id: str, index: int) -> AnswerAssessment:
    path = answer_result_path(root, session_id, index)
    return _answer_from_json(json.loads(path.read_text(encoding="utf-8")))


def read_all_answers(
    root: Path, session_id: str, questions: Sequence[QuestionRecord]
) -> tuple[AnswerAssessment, ...]:
    """Every answer already judged, in question order. Missing ones are skipped."""
    found = []
    for record in questions:
        if answer_result_path(root, session_id, record.index).exists():
            found.append(read_answer(root, session_id, record.index))
    return tuple(found)


def aggregate_available(
    questions: Sequence[QuestionRecord],
    assessments: Sequence[AnswerAssessment],
) -> ExamResult:
    """Aggregate over the answers that exist.

    `core.aggregate` requires the assessments to match the blueprint exactly, so
    a failed answer would make it raise. Passing only the slots that were judged
    keeps that check meaningful — it still catches a mismatch — while letting an
    exam with a broken answer produce a level.
    """
    judged = {a.question_index for a in assessments}
    blueprint = tuple(
        question_from_record(record).slot
        for record in questions
        if record.index in judged
    )
    return aggregate(blueprint, list(assessments))


def result_json(result: ExamResult, expected: int) -> dict:
    return {
        "final_level": result.final_level.name,
        "scored_answers": len(result.assessments),
        "expected_answers": expected,
        "dimensions": {d.value: level.name for d, level in result.dimension_levels},
        "by_type": {t.value: level.name for t, level in result.by_type},
        "by_topic": {topic: level.name for topic, level in result.by_topic},
        "pronunciation": {
            "mean_confidence": result.pronunciation.mean_confidence,
            "low_confidence_rate": result.pronunciation.low_confidence_rate,
            "indicative": result.pronunciation.indicative,
        },
        "answers": [_answer_json(a) for a in result.assessments],
    }


def _answer_json(assessment: AnswerAssessment) -> dict:
    metrics = assessment.metrics
    return {
        "index": assessment.question_index,
        "question_en": assessment.question.text_en,
        "slot": {
            "qtype": assessment.question.slot.qtype.value,
            "topic_id": assessment.question.slot.topic_id,
            "combo_id": assessment.question.slot.combo_id,
            "advanced": assessment.question.slot.advanced,
            "target": assessment.question.slot.target.name,
        },
        "transcript": {
            "text": assessment.transcript.text,
            "duration_s": assessment.transcript.duration_s,
            "words": [
                {"text": w.text, "start": w.start, "end": w.end, "probability": w.probability}
                for w in assessment.transcript.words
            ],
            "speech_spans": [
                {"start": s.start, "end": s.end} for s in assessment.transcript.speech_spans
            ],
        },
        "metrics": {
            "duration_s": metrics.duration_s,
            "word_count": metrics.word_count,
            "words_per_minute": metrics.words_per_minute,
            "pause_ratio": metrics.pause_ratio,
            "type_token_ratio": metrics.type_token_ratio,
            "mean_utterance_length": metrics.mean_utterance_length,
            "mean_confidence": metrics.mean_confidence,
            "low_confidence_rate": metrics.low_confidence_rate,
        },
        "scores": [
            {
                "dimension": s.dimension.value,
                "level": s.level.name,
                "comment_ko": s.comment_ko,
                "evidence": [
                    {"quote": e.quote, "issue_ko": e.issue_ko, "correction_en": e.correction_en}
                    for e in s.evidence
                ],
            }
            for s in assessment.dimension_scores
        ],
    }


def _answer_from_json(payload: dict) -> AnswerAssessment:
    slot = payload["slot"]
    transcript = payload["transcript"]
    metrics = payload["metrics"]
    return AnswerAssessment(
        question=Question(
            slot=QuestionSlot(
                index=payload["index"],
                qtype=QuestionType(slot["qtype"]),
                topic_id=slot["topic_id"],
                combo_id=slot["combo_id"],
                advanced=slot["advanced"],
                target=ActflLevel[slot["target"]],
            ),
            text_en=payload["question_en"],
        ),
        transcript=Transcript(
            text=transcript["text"],
            words=tuple(
                Word(w["text"], w["start"], w["end"], w["probability"])
                for w in transcript["words"]
            ),
            speech_spans=tuple(
                Span(s["start"], s["end"]) for s in transcript["speech_spans"]
            ),
            duration_s=transcript["duration_s"],
        ),
        dimension_scores=tuple(
            DimensionScore(
                dimension=Dimension(s["dimension"]),
                level=ActflLevel[s["level"]],
                comment_ko=s["comment_ko"],
                evidence=tuple(
                    Evidence(e["quote"], e["issue_ko"], e["correction_en"])
                    for e in s["evidence"]
                ),
            )
            for s in payload["scores"]
        ),
        metrics=AnswerMetrics(**metrics),
    )
```

```python
class ExplodingRater:
    """Fails a fixed number of times, then succeeds — or never, if `always`."""

    def __init__(self, failures: int, always: bool = False) -> None:
        self.failures = failures
        self.always = always
        self.attempts = 0

    def rate(self, question, transcript, metrics):
        self.attempts += 1
        if self.always or self.attempts <= self.failures:
            raise TimeoutError("ollama timed out")
        return FakeRater().rate(question, transcript, metrics)
```

```python
# src/opic/session/worker.py
"""One queue, one thread, four kinds of task.

Sharing a queue between question generation and answer evaluation is not
laziness: it is what guarantees that two requests never hit Ollama at once. The
rater measurement in `notes/2026-08-19-rater-model.md` shows what concurrent
pressure on a local model costs — four timeouts out of five.

Nothing here is kept in memory that is not also on disk, so `resume()` after a
restart needs no state beyond what the session folder already says.
"""

from __future__ import annotations

import queue
import threading
import time
from collections.abc import Callable
from dataclasses import dataclass
from pathlib import Path

from opic.core.metrics import compute_metrics
from opic.core.questions import build_blueprint
from opic.core.scoring import AnswerAssessment
from opic.session import assessments as answers_on_disk
from opic.session import store
from opic.session.state import (
    AnswerRecord,
    AnswerState,
    QuestionRecord,
    SessionState,
    Step,
    advance,
)

GENERATE = "generate"
SYNTHESIZE = "synthesize"
EVALUATE = "evaluate"
AGGREGATE = "aggregate"


@dataclass(frozen=True, slots=True)
class Task:
    kind: str
    session_id: str
    index: int | None = None


class Worker:
    def __init__(
        self,
        root: Path,
        tts,
        stt,
        writer,
        rater,
        retries: int = 3,
        sleeper: Callable[[float], None] = time.sleep,
    ) -> None:
        self._root = root
        self._tts = tts
        self._stt = stt
        self._writer = writer
        self._rater = rater
        self._retries = retries
        self._sleep = sleeper
        self._queue: queue.Queue[Task | None] = queue.Queue()
        self._lock = threading.Lock()
        self._thread: threading.Thread | None = None

    # --- state access -----------------------------------------------------

    def update(
        self, session_id: str, change: Callable[[SessionState], SessionState]
    ) -> SessionState:
        """Read, change and write a session under the lock.

        Both the HTTP thread and the worker thread mutate sessions, and every
        write rewrites the whole file. Serialising them here is what stops one
        from overwriting the other's change.
        """
        with self._lock:
            state = change(store.load(self._root, session_id))
            store.save(self._root, state)
            return state

    # --- queue ------------------------------------------------------------

    def submit(self, task: Task) -> None:
        self._queue.put(task)

    def drain(self) -> None:
        """Run every queued task in this thread. Used by tests and by `resume`."""
        while True:
            try:
                task = self._queue.get_nowait()
            except queue.Empty:
                return
            if task is not None:
                self.run_one(task)

    def start(self) -> None:
        if self._thread is not None:
            return
        self._thread = threading.Thread(target=self._loop, daemon=True, name="opic-worker")
        self._thread.start()

    def stop(self) -> None:
        if self._thread is None:
            return
        self._queue.put(None)
        self._thread.join(timeout=5)
        self._thread = None

    def _loop(self) -> None:
        while True:
            task = self._queue.get()
            if task is None:
                return
            try:
                self.run_one(task)
            except Exception:  # noqa: BLE001 - a dead worker is worse than a logged failure
                import traceback

                traceback.print_exc()

    # --- tasks ------------------------------------------------------------

    def run_one(self, task: Task) -> None:
        if task.kind == GENERATE:
            self._generate(task.session_id)
        elif task.kind == SYNTHESIZE:
            self._synthesize(task.session_id, task.index)
        elif task.kind == EVALUATE:
            self._evaluate(task.session_id, task.index)
        elif task.kind == AGGREGATE:
            self._aggregate(task.session_id)
        else:
            raise ValueError(f"unknown task kind {task.kind!r}")

    def _generate(self, session_id: str) -> None:
        state = store.load(self._root, session_id)
        blueprint = build_blueprint(state.survey, state.self_assess_level, state.seed)
        written = self._writer.write(state.survey, blueprint)
        records = tuple(
            QuestionRecord(
                index=question.slot.index,
                qtype=question.slot.qtype.value,
                topic_id=question.slot.topic_id,
                combo_id=question.slot.combo_id,
                advanced=question.slot.advanced,
                target=question.slot.target.name,
                text_en=question.text_en,
                audio_path=None,
            )
            for question in written
        )
        self.update(session_id, lambda s: s.with_questions(records))
        for record in records:
            self.submit(Task(SYNTHESIZE, session_id, record.index))

    def _synthesize(self, session_id: str, index: int) -> None:
        state = store.load(self._root, session_id)
        record = state.questions[index]
        if record.audio_path:
            return
        audio = self._tts.synthesize(record.text_en)
        self.update(session_id, lambda s: s.with_question_audio(index, str(audio)))

    def _evaluate(self, session_id: str, index: int) -> None:
        state = store.load(self._root, session_id)
        record = state.answers[index]
        if record.state in (AnswerState.SCORED, AnswerState.FAILED):
            return
        audio = store.session_dir(self._root, session_id) / record.audio_path

        try:
            assessment = self._judge(state, index, audio)
        except Exception as error:  # noqa: BLE001 - one bad answer must not end the exam
            reason = f"{type(error).__name__}: {error}"
            self.update(
                session_id,
                lambda s: s.with_answer(
                    AnswerRecord(index, AnswerState.FAILED, record.audio_path, reason)
                ),
            )
        else:
            answers_on_disk.write_answer(self._root, session_id, assessment)
            self.update(
                session_id,
                lambda s: s.with_answer(
                    AnswerRecord(index, AnswerState.SCORED, record.audio_path)
                ),
            )

        after = store.load(self._root, session_id)
        settled = sum(
            1
            for a in after.answers
            if a.state in (AnswerState.SCORED, AnswerState.FAILED)
        )
        if settled == len(after.questions):
            self.submit(Task(AGGREGATE, session_id))

    def _judge(self, state: SessionState, index: int, audio: Path) -> AnswerAssessment:
        question = answers_on_disk.question_from_record(state.questions[index])
        transcript = self._attempt(lambda: self._stt.transcribe(audio))
        metrics = compute_metrics(transcript)
        scores = self._attempt(lambda: self._rater.rate(question, transcript, metrics))
        return AnswerAssessment(
            question=question,
            transcript=transcript,
            dimension_scores=scores,
            metrics=metrics,
        )

    def _attempt(self, call):
        """Retry a model call, backing off, before giving up on this answer."""
        last: Exception | None = None
        for attempt in range(self._retries):
            try:
                return call()
            except Exception as error:  # noqa: BLE001 - retried, then surfaced
                last = error
                self._sleep(2**attempt)
        raise last

    def _aggregate(self, session_id: str) -> None:
        state = store.load(self._root, session_id)
        judged = answers_on_disk.read_all_answers(self._root, session_id, state.questions)
        if not judged:
            return
        result = answers_on_disk.aggregate_available(state.questions, judged)
        payload = answers_on_disk.result_json(result, expected=len(state.questions))
        target = store.result_path(self._root, session_id)
        temporary = target.with_suffix(".json.tmp")
        import json

        temporary.write_text(
            json.dumps(payload, ensure_ascii=False, indent=2), encoding="utf-8"
        )
        temporary.replace(target)
        if state.step is Step.GRADING:
            self.update(session_id, lambda s: advance(s, Step.DONE))

    # --- restart ----------------------------------------------------------

    def resume(self) -> None:
        """Requeue whatever every unfinished session still owes."""
        for session_id in store.list_unfinished(self._root):
            state = store.load(self._root, session_id)
            if state.self_assess_level is None:
                continue
            if not state.questions:
                self.submit(Task(GENERATE, session_id))
                continue
            for question in state.questions:
                if not question.audio_path:
                    self.submit(Task(SYNTHESIZE, session_id, question.index))
            for answer in state.answers:
                if answer.state is AnswerState.RECORDED:
                    self.submit(Task(EVALUATE, session_id, answer.index))
```

```python
"""HTTP only: validate, delegate to `session/`, answer JSON.

No exam logic lives here. Anything that decides what happens next belongs in
`opic.session`, which is why these handlers read as plumbing.
"""

from __future__ import annotations

from collections.abc import Callable
from datetime import datetime
from pathlib import Path

from fastapi import FastAPI, HTTPException
from fastapi.responses import JSONResponse
from pydantic import BaseModel

from opic.core.survey import (
    GROUP_LABELS_KO,
    GROUP_MINIMUMS,
    HOUSING_LABELS_KO,
    MIN_TOTAL_ITEMS,
    OCCUPATION_LABELS_KO,
    RECENT_COURSE_LABELS_KO,
    SURVEY_ITEMS,
    Housing,
    Occupation,
    RecentCourse,
    SurveyAnswers,
    TopicGroup,
    validate_survey,
)
from opic.session import store
from opic.session.state import Step, advance
from opic.session.worker import Worker


class SurveyPayload(BaseModel):
    occupation: str
    is_student: bool
    housing: str
    selected_item_ids: list[str]
    recent_course: str | None = None


def create_app(
    root: Path,
    worker: Worker,
    clock: Callable[[], datetime] = datetime.now,
) -> FastAPI:
    app = FastAPI(title="OPIc simulator")

    def session_json(state) -> dict:
        payload = store.to_json(state)
        payload["questions_ready"] = state.questions_ready
        return payload

    def load_or_404(session_id: str):
        try:
            return store.load(root, session_id)
        except FileNotFoundError:
            raise HTTPException(status_code=404, detail="session not found") from None

    @app.post("/api/sessions", status_code=201)
    def create_session() -> dict:
        state = store.create(root, seed=int(clock().timestamp()), now=clock())
        return session_json(state)

    @app.get("/api/sessions/{session_id}")
    def read_session(session_id: str) -> dict:
        return session_json(load_or_404(session_id))

    @app.get("/api/survey")
    def survey_schema() -> dict:
        return {
            "min_total_items": MIN_TOTAL_ITEMS,
            "occupations": [
                {"id": occupation.value, "label_ko": label}
                for occupation, label in OCCUPATION_LABELS_KO.items()
            ],
```

```javascript
// The whole exam is one page. This module owns the session id, talks to the
// API, and hands the current state to whichever step module renders it.
const steps = {};

export const api = {
  async get(path) {
    const response = await fetch(path);
    if (!response.ok) throw Object.assign(new Error(path), {status: response.status});
    return response.json();
  },
  async send(method, path, body) {
    const response = await fetch(path, {
      method,
      headers: {"Content-Type": "application/json"},
      body: JSON.stringify(body),
    });
    const payload = await response.json().catch(() => ({}));
    if (!response.ok) throw Object.assign(new Error(path), {status: response.status, payload});
    return payload;
  },
  async upload(path, blob, filename) {
    const form = new FormData();
    form.append("audio", blob, filename);
    const response = await fetch(path, {method: "POST", body: form});
    if (!response.ok) throw Object.assign(new Error(path), {status: response.status});
    return response.json();
  },
};

const LABELS = {
  survey: "1단계 · 배경 설문",
  level: "2단계 · 자기 평가",
  setup: "3단계 · 마이크 점검",
  sample: "4단계 · 예시 문항",
  test: "5단계 · 시험",
  grading: "채점 중",
  done: "결과",
};

export const root = document.getElementById("app");

export async function show(state) {
  document.getElementById("step-label").textContent = LABELS[state.step] ?? "";
  const step = steps[state.step];
  if (!step) {
    root.textContent = `알 수 없는 단계: ${state.step}`;
    return;
  }
  root.replaceChildren();
  await step.render(state, {show, api});
}

export async function reload(sessionId) {
  return show(await api.get(`/api/sessions/${sessionId}`));
}

async function boot() {
  const stored = localStorage.getItem("opic.session");
  let state = null;
  if (stored) {
    try {
      state = await api.get(`/api/sessions/${stored}`);
    } catch (error) {
      localStorage.removeItem("opic.session");
    }
  }
  if (!state) {
    state = await api.send("POST", "/api/sessions", {});
    localStorage.setItem("opic.session", state.id);
  }
  await show(state);
}

for (const name of ["survey", "level", "setup", "sample", "test", "grading", "done"]) {
  try {
```

```javascript
// Microphone access and recording, in one place: the Setup step and the Test
// step must behave identically, and the browser's supported MIME type is
// decided once, here.
const CANDIDATES = ["audio/webm;codecs=opus", "audio/webm", "audio/mp4"];

export function supportedType() {
  return CANDIDATES.find((type) => MediaRecorder.isTypeSupported(type)) ?? "";
}

export async function askForMicrophone() {
  try {
    return await navigator.mediaDevices.getUserMedia({audio: true});
  } catch (error) {
    throw new Error("마이크 권한이 필요합니다. 브라우저 설정에서 허용해 주세요.");
  }
}

export function startRecording(stream) {
  const type = supportedType();
  const recorder = new MediaRecorder(stream, type ? {mimeType: type} : undefined);
  const chunks = [];
  recorder.ondataavailable = (event) => {
    if (event.data.size > 0) chunks.push(event.data);
  };
  recorder.start();

  return {
    stop() {
      return new Promise((resolve) => {
        recorder.onstop = () => {
          const blob = new Blob(chunks, {type: recorder.mimeType});
          const extension = recorder.mimeType.includes("mp4") ? "mp4" : "webm";
          resolve({blob, filename: `answer.${extension}`});
        };
        recorder.stop();
      });
    },
  };
}
```

```javascript
// The mechanics of one question: play it, record the answer, hand back the
// blob. Both the sample question and the real test run this, so the sample is
// a genuine rehearsal and not a different code path.
import {askForMicrophone, startRecording} from "./recorder.js";

const POLL_MS = 3000;

export async function waitUntilReady(api, sessionId) {
  let state = await api.get(`/api/sessions/${sessionId}`);
  while (!state.questions_ready) {
    await new Promise((resolve) => setTimeout(resolve, POLL_MS));
    state = await api.get(`/api/sessions/${sessionId}`);
  }
  return state;
}

export function askAndRecord({audioUrl, onState}) {
  // Returns {play, replay, stop}: the caller owns the buttons, this owns the
  // state machine. `replay` is allowed exactly once, as in the real exam.
  let stream = null;
  let session = null;
  let replays = 0;
  const audio = new Audio(audioUrl);

  async function play() {
    onState("playing");
    await audio.play();
    await new Promise((resolve) => audio.addEventListener("ended", resolve, {once: true}));
    stream = await askForMicrophone();
    session = startRecording(stream);
    onState("recording");
  }

  async function replay() {
    if (replays >= 1) return false;
    replays += 1;
    audio.currentTime = 0;
    await audio.play();
    return true;
  }

  async function stop() {
    const recorded = await session.stop();
    stream.getTracks().forEach((track) => track.stop());
    onState("uploading");
    return recorded;
  }

  return {play, replay, stop, replaysLeft: () => 1 - replays};
}

export async function uploadWithRetries(api, path, blob, filename, attempts = 3) {
  let last = null;
  for (let attempt = 0; attempt < attempts; attempt += 1) {
    try {
      return await api.upload(path, blob, filename);
    } catch (error) {
      last = error;
      await new Promise((resolve) => setTimeout(resolve, 500 * 2 ** attempt));
    }
  }
  throw last;
}
```

```javascript
    const settled = progress.scored + progress.failed;
    bar.textContent = `채점 완료: ${settled} / ${progress.expected}`;
    const left = Math.max(0, progress.expected - settled) * SECONDS_PER_ANSWER;
    eta.textContent = `남은 예상 시간: 약 ${Math.ceil(left / 60)}분`;
    failures.textContent =
      progress.failed > 0 ? `채점하지 못한 답변: ${progress.failed}개` : "";

    if (progress.done) {
      await show(await api.get(`/api/sessions/${state.id}`));
      return;
    }
    if (settled === progress.expected) {
      if (progress.scored === 0) {
        // Nothing was scored, so no result will ever be written. Saying so
        // beats a spinner that never stops.
        heading.textContent = "채점하지 못했습니다";
        eta.textContent = "";
        failures.textContent =
          "모든 답변의 채점에 실패했습니다. Ollama가 실행 중인지 확인한 뒤 서버를 다시 시작하면 남은 답변부터 이어서 채점합니다.";
        return;
      }
      eta.textContent = "결과를 정리하는 중…";
    }
    await new Promise((resolve) => setTimeout(resolve, POLL_MS));
  }
}
```

```python
`uv run uvicorn --factory opic.web.serve:app` is the supported way to start it.
Two uvicorn workers would be two queues fighting over one model and one set of

from opic.adapters.tts import SayTextToSpeech

from opic.web.app import create_app

SESSIONS = Path("data/sessions")
TTS_CACHE = Path("data/tts_cache")


def app():

    worker = Worker(
        root=SESSIONS,

        stt=WhisperSpeechToText(),
```

```python
import os

LOCK_NAME = ".server.lock"


class LockedError(RuntimeError):
    """Another server already owns this sessions folder."""


def _is_running(pid: int) -> bool:
    try:
        os.kill(pid, 0)
    except (ProcessLookupError, ValueError):
        return False
    except PermissionError:
        return True
    return True


def claim_the_lock(root: Path, pid: int | None = None, is_running=_is_running) -> None:
    root.mkdir(parents=True, exist_ok=True)
    lock = root / LOCK_NAME
    if lock.exists():
        held = lock.read_text(encoding="utf-8").strip()
        if held.isdigit() and is_running(int(held)):
            raise LockedError(
                f"a server is already running on {root} (pid {held}); "
                "evaluation must not run twice over the same sessions"
            )
    lock.write_text(str(pid or os.getpid()), encoding="utf-8")
```

```python
def run_exam(
    answers: SurveyAnswers,
    self_assess_level: int,
    seed: int,
    session_dir: Path,
    tts,
    stt,
    writer,
    rater,
    recorder: Callable[[Path, int, str], Path] = record_answer,
    player: Callable[[Path], None] = play,
    device: str = ":0",
) -> ExamResult:
    """Run a whole exam over the same session machinery the web interface uses.

    `session_dir` keeps meaning what it meant: this session's folder. It is
    split into the sessions root and the session id so `store` and `Worker` —
    which think in those terms — can be handed the exact layout the CLI already
    wrote.
    """
    issues = validate_survey(answers)
    if issues:
        detail = ", ".join(f"{i.error.value}({i.detail})" for i in issues)
        raise ValueError(f"invalid survey: {detail}")

    root = session_dir.parent
    session_id = session_dir.name
    worker = Worker(root=root, tts=tts, stt=stt, writer=writer, rater=rater)

    session_dir.mkdir(parents=True, exist_ok=True)
    state = (
        SessionState.new(session_id, seed)
        .with_survey(answers)
        .with_level(self_assess_level)
    )
    store.save(root, state)

    worker.submit(Task("generate", session_id))
    worker.drain()

    state = store.load(root, session_id)
    print(f"\n총 {len(state.questions)}개의 질문이 있습니다.\n")

    for question in state.questions:
        print(f"질문 {question.index + 1} / {len(state.questions)}")
        player(Path(question.audio_path))
        recording = store.answer_audio_path(root, session_id, question.index, ".wav")
        recorder(recording, ANSWER_SECONDS, device)
        worker.update(
            session_id,
            lambda s, i=question.index, name=recording.name: s.with_answer(
                AnswerRecord(i, AnswerState.RECORDED, name)
            ),
        )
        worker.submit(Task("evaluate", session_id, question.index))
        print(f"  녹음 완료: {ANSWER_SECONDS}초")

    print("\n채점 중입니다. 15개 답변에 약 1시간이 걸립니다.\n")
    worker.drain()

    state = store.load(root, session_id)
    judged = assessments.read_all_answers(root, session_id, state.questions)
    if not judged:
        raise RuntimeError("no answer could be scored; is Ollama running?")
    print(f"  채점 {len(judged)} / {len(state.questions)}")
    return assessments.aggregate_available(state.questions, judged)
```
